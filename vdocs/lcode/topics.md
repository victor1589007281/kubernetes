### 代码模板

```go
// 通用状态机框架 (k次交易)
func maxProfit(k int, prices []int) int {
    n := len(prices)
    if n == 0 { return 0 }
    
    // dp[i][j][0/1] = 第i天，完成j次交易，不持有/持有 的最大利润
    dp := make([][][2]int, n)
    for i := range dp {
        dp[i] = make([][2]int, k+1)
    }
    
    // 初始化
    for j := 0; j <= k; j++ {
        dp[0][j][1] = -prices[0]
    }
    
    for i := 1; i < n; i++ {
        for j := 1; j <= k; j++ {
            dp[i][j][0] = max(dp[i-1][j][0], dp[i-1][j][1]+prices[i])
            dp[i][j][1] = max(dp[i-1][j][1], dp[i-1][j-1][0]-prices[i])
        }
    }
    return dp[n-1][k][0]
}
```

```go
func maxPathSum(root *TreeNode) int {
    maxSum := math.MinInt32
    
    var dfs func(node *TreeNode) int
    dfs = func(node *TreeNode) int {
        if node == nil { return 0 }
        
        left := max(0, dfs(node.Left))   // 负数贡献为0
        right := max(0, dfs(node.Right))
        
        // 当前节点作为拐点
        maxSum = max(maxSum, left + right + node.Val)
        
        // 向上贡献的最大值
        return max(left, right) + node.Val
    }
    
    dfs(root)
    return maxSum
}
```

### 单调栈原理

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                          单调递减栈                                          │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│   找下一个更大的元素                                                         │
│                                                                              │
│   nums = [2, 1, 2, 4, 3]                                                    │
│                                                                              │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  从右向左遍历，维护递减栈                                          │    │
│   │                                                                    │    │
│   │  i=4: 3 入栈, stack=[3], ans[4]=-1 (栈中无更大)                   │    │
│   │  i=3: 4>3, 弹出3, stack=[], 4入栈, ans[3]=-1                      │    │
│   │  i=2: 2<4, 栈顶4就是答案, stack=[4,2], ans[2]=4                   │    │
│   │  i=1: 1<2, 栈顶2就是答案, stack=[4,2,1], ans[1]=2                 │    │
│   │  i=0: 2>1, 弹出1, 2=2不大于, 栈顶4是答案, stack=[4,2,2], ans[0]=4 │    │
│   │                                                                    │    │
│   │  ans = [4, 2, 4, -1, -1]                                          │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
│   为什么有效?                                                                │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  栈中元素递减，栈顶就是当前元素右边第一个更大的                    │    │
│   │  被弹出的元素不可能是后续元素的"下一个更大"(当前元素更合适)       │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
```

### 区间贪心原理

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                          区间贪心策略                                        │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│   问题: 选择最多不重叠区间 / 删除最少使不重叠                                │
│                                                                              │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  策略: 按结束时间排序，优先选择结束早的                            │    │
│   │                                                                    │    │
│   │  原理: 结束越早，留给后面的空间越多                                │    │
│   │                                                                    │    │
│   │  ───────────────────────────────────────▶ 时间轴                  │    │
│   │  ├──────┤                               区间A (结束早)             │    │
│   │       ├─────────────┤                   区间B                      │    │
│   │            ├────┤                       区间C                      │    │
│   │                                                                    │    │
│   │  选A比选B好，因为A结束后还能选C                                    │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
│   代码模板:                                                                  │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  sort.Slice(intervals, func(i, j int) bool {                       │    │
│   │      return intervals[i][1] < intervals[j][1] // 按结束时间排序    │    │
│   │  })                                                                 │    │
│   │                                                                    │    │
│   │  count := 1                                                        │    │
│   │  end := intervals[0][1]                                            │    │
│   │  for i := 1; i < len(intervals); i++ {                             │    │
│   │      if intervals[i][0] >= end {  // 不重叠                        │    │
│   │          count++                                                   │    │
│   │          end = intervals[i][1]                                     │    │
│   │      }                                                              │    │
│   │  }                                                                  │    │
│   │  return len(intervals) - count  // 需要删除的数量                  │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
```
