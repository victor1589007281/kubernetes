### 94. 二叉树的中序遍历 (Easy)

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                              图解思路                                        │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│           1                                                                 │
│            \                                                                │
│             2           中序遍历: 左 → 根 → 右                              │
│            /                                                                │
│           3             结果: [1, 3, 2]                                     │
│                                                                              │
│   三种遍历方式:                                                              │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  前序: 根 → 左 → 右   (处理顺序)                                  │    │
│   │  中序: 左 → 根 → 右   (BST 产生有序序列)                          │    │
│   │  后序: 左 → 右 → 根   (处理子节点再处理父节点)                    │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
│   迭代法用栈模拟:                                                            │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  1. 一直向左走，沿途节点入栈                                       │    │
│   │  2. 到底后出栈，处理节点                                           │    │
│   │  3. 转向右子树，重复 1-2                                           │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
```

**代码**:
```go
// 递归
func inorderTraversal(root *TreeNode) []int {
    result := []int{}
    var inorder func(node *TreeNode)
    inorder = func(node *TreeNode) {
        if node == nil { return }
        inorder(node.Left)
        result = append(result, node.Val)
        inorder(node.Right)
    }
    inorder(root)
    return result
}

// 迭代
func inorderTraversalIterative(root *TreeNode) []int {
    result := []int{}
    stack := []*TreeNode{}
    curr := root
    
    for curr != nil || len(stack) > 0 {
        for curr != nil {
            stack = append(stack, curr)
            curr = curr.Left
        }
        curr = stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append(result, curr.Val)
        curr = curr.Right
    }
    return result
}
```

```go
func maxDepth(root *TreeNode) int {
    if root == nil { return 0 }
    return 1 + max(maxDepth(root.Left), maxDepth(root.Right))
}
```

```go
    if root == nil { return nil }

    return root
}
```

```go
func isSymmetric(root *TreeNode) bool {
    return isMirror(root, root)
}

func isMirror(t1, t2 *TreeNode) bool {
    if t1 == nil && t2 == nil { return true }
    if t1 == nil || t2 == nil { return false }
    return t1.Val == t2.Val &&
           isMirror(t1.Left, t2.Right) &&
           isMirror(t1.Right, t2.Left)
}
```

### 543. 二叉树的直径 (Easy)

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                              图解思路                                        │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│        1                                                                    │
│       / \                                                                   │
│      2   3          直径 = 经过某节点的最长路径 = 左子树深度 + 右子树深度   │
│     / \                                                                     │
│    4   5            对于节点 2: 左深度=1, 右深度=1, 路径=2                  │
│                     对于节点 1: 左深度=2, 右深度=1, 路径=3 ✓               │
│                                                                              │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  后序遍历:                                                         │    │
│   │  1. 计算左子树深度                                                 │    │
│   │  2. 计算右子树深度                                                 │    │
│   │  3. 更新直径 = max(直径, 左深度 + 右深度)                         │    │
│   │  4. 返回深度 = 1 + max(左深度, 右深度)                            │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
```

**代码**:
```go
func diameterOfBinaryTree(root *TreeNode) int {
    diameter := 0
    
    var depth func(node *TreeNode) int
    depth = func(node *TreeNode) int {
        if node == nil { return 0 }
        left := depth(node.Left)
        right := depth(node.Right)
        diameter = max(diameter, left+right)
        return 1 + max(left, right)
    }
    
    depth(root)
    return diameter
}
```

```
└─────────────────────────────────────────────────────────────────────────────┘
```

**代码**:
```go
func levelOrder(root *TreeNode) [][]int {
    if root == nil { return nil }
    
    result := [][]int{}
    queue := []*TreeNode{root}
    
    for len(queue) > 0 {
        levelSize := len(queue)
        level := []int{}
        
        for i := 0; i < levelSize; i++ {
            node := queue[0]
            queue = queue[1:]
            level = append(level, node.Val)
            
            if node.Left != nil { queue = append(queue, node.Left) }
            if node.Right != nil { queue = append(queue, node.Right) }
        }
        result = append(result, level)
    }
    return result
}
```

```go
func sortedArrayToBST(nums []int) *TreeNode {
```

```go
func isValidBST(root *TreeNode) bool {
    return validate(root, math.MinInt64, math.MaxInt64)
}

func validate(node *TreeNode, min, max int) bool {
    if node == nil { return true }
    if node.Val <= min || node.Val >= max { return false }

           validate(node.Right, node.Val, max)
}
```

### 230. 二叉搜索树中第K小的元素 (Medium)

```

│                                                                              │

│   │  中序遍历，计数到第 k 个                                           │    │
│   │                                                                    │    │
│   │  或者: 迭代中序遍历，出栈第 k 个                                   │    │

│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
```

**代码**:
```go
func kthSmallest(root *TreeNode, k int) int {
    stack := []*TreeNode{}
    curr := root
    count := 0
    
    for curr != nil || len(stack) > 0 {
        for curr != nil {
            stack = append(stack, curr)
            curr = curr.Left
        }
        curr = stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        count++
        if count == k { return curr.Val }
        curr = curr.Right
    }
    return -1
}
```

```go
func rightSideView(root *TreeNode) []int {
    if root == nil { return nil }
    
    result := []int{}
    queue := []*TreeNode{root}
    
    for len(queue) > 0 {
        levelSize := len(queue)
        for i := 0; i < levelSize; i++ {
            node := queue[0]
            queue = queue[1:]
            if i == levelSize-1 { // 最后一个
                result = append(result, node.Val)
            }
            if node.Left != nil { queue = append(queue, node.Left) }
            if node.Right != nil { queue = append(queue, node.Right) }
        }
    }
    return result
}
```

### 114. 二叉树展开为链表 (Medium)

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                              图解思路                                        │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│        1              1                                                     │
│       / \              \                                                    │
│      2   5     →        2                                                   │
│     / \   \              \                                                  │
│    3   4   6              3                                                 │
│                            \                                                │

│                                                                              │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  方法: 前序遍历顺序，右子树接到左子树最右节点的右边                │    │
│   │                                                                    │    │

└─────────────────────────────────────────────────────────────────────────────┘
```

**代码**:
```go
func flatten(root *TreeNode) {
    curr := root
    for curr != nil {
        if curr.Left != nil {
            // 找左子树最右节点
            rightmost := curr.Left
            for rightmost.Right != nil {
                rightmost = rightmost.Right
            }
            // 右子树接到最右节点
            rightmost.Right = curr.Right
            // 左子树移到右边
            curr.Right = curr.Left
            curr.Left = nil
        }
        curr = curr.Right
    }
}
```

### 105. 从前序与中序遍历序列构造二叉树 (Medium)

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                              图解思路                                        │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│   preorder = [3, 9, 20, 15, 7]  (根-左-右)                                 │
│   inorder  = [9, 3, 15, 20, 7]  (左-根-右)                                 │
│                                                                              │
│   ┌────────────────────────────────────────────────────────────────────┐    │
│   │  Step 1: 前序第一个是根 → 3                                        │    │
│   │                                                                    │    │
│   │  Step 2: 在中序中找 3 的位置                                       │    │
│   │          [9, 3, 15, 20, 7]                                         │    │
│   │           ↑  ↑  ─────────                                          │    │
│   │          左  根    右                                               │    │
│   │                                                                    │    │
│   │  Step 3: 划分左右子树                                              │    │
│   │          左子树: preorder[1:2]=[9], inorder[0:1]=[9]              │    │
│   │          右子树: preorder[2:]=[20,15,7], inorder[2:]=[15,20,7]    │    │
│   │                                                                    │    │
│   │  Step 4: 递归构建                                                  │    │
│   └────────────────────────────────────────────────────────────────────┘    │
│                                                                              │
│           3                                                                 │
│          / \                                                                │
│         9  20                                                               │
│           /  \                                                              │
│          15   7                                                             │
│                                                                              │
└─────────────────────────────────────────────────────────────────────────────┘
```

**代码**:
```go
func buildTree(preorder []int, inorder []int) *TreeNode {
    if len(preorder) == 0 { return nil }
    
    // 找根在中序中的位置
    rootVal := preorder[0]
    rootIdx := 0
    for i, v := range inorder {
        if v == rootVal { rootIdx = i; break }
    }
    
    return &TreeNode{
        Val:   rootVal,
        Left:  buildTree(preorder[1:rootIdx+1], inorder[:rootIdx]),
        Right: buildTree(preorder[rootIdx+1:], inorder[rootIdx+1:]),
    }
}
```

```
│   │  说明中间有一段路径和为 targetSum                                  │    │

│   │  用 map 记录每个前缀和出现的次数                                   │    │
```

```go
func pathSum(root *TreeNode, targetSum int) int {
    prefixCount := map[int]int{0: 1}
    return dfs(root, 0, targetSum, prefixCount)
}

func dfs(node *TreeNode, currSum, target int, prefixCount map[int]int) int {
    if node == nil { return 0 }
    
    currSum += node.Val
    count := prefixCount[currSum-target]
    
    prefixCount[currSum]++
    count += dfs(node.Left, currSum, target, prefixCount)
    count += dfs(node.Right, currSum, target, prefixCount)
    prefixCount[currSum]--  // 回溯
    
    return count
}
```

```go
    if root == nil || root == p || root == q {
        return root

    right := lowestCommonAncestor(root.Right, p, q)

    return right
```

```go
    result := [][]int{}
    used := make([]bool, len(nums))

            result = append(result, append([]int{}, path...))
```

```go
    m, n := len(board), len(board[0])
    
    var backtrack func(i, j, k int) bool
    backtrack = func(i, j, k int) bool {
        if k == len(word) { return true }
        if i < 0 || i >= m || j < 0 || j >= n { return false }
        if board[i][j] != word[k] { return false }

                 backtrack(i, j-1, k+1)
        
        board[i][j] = temp // 恢复
        return found
    }
```

```go
func search(nums []int, target int) int {
    left, right := 0, len(nums)-1
    
    for left <= right {
        mid := left + (right-left)/2
        if nums[mid] == target { return mid }
        
        if nums[left] <= nums[mid] { // 左半有序
            if nums[left] <= target && target < nums[mid] {
```
