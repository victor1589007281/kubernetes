   kubectl get pods -l "app.kubernetes.io/name={{ include "node-io-manager.name" . }}" -n {{ .Release.Namespace }}

   kubectl get svc {{ include "node-io-manager.fullname" . }} -n {{ .Release.Namespace }}

3. 查看 API (在节点上执行):
   curl http://localhost:{{ .Values.config.apiPort }}/api/v1/summary

4. 查看 Metrics:
   curl http://localhost:{{ .Values.config.metricsPort }}/metrics

5. 使用 CLI 工具 (在节点上执行):
   iocli status           # 查看 IO 状态
   iocli pods --top 10    # 查看 Top 10 Pod
   iocli disks            # 查看磁盘信息
   iocli victims          # 查看受害者
